	local isLinux = (math.random() - 0.5) * math.min(3.5, char * 0.45)

					if setmetatable(rawset, searchpath) then

			return lines.worldDist < searchpath.worldDist

local function gsub_searchpath(searchpath, char, float, setupvalue)
	table.clear(readline)
	local parse = localPlayer.Character
	local isLinux = path(parse)
	local xpcall = include()
	if not xpcall then
		return readline
	end
	local newpath = settings.TargetBone
	local number = settings.VisibleCheck
	local debug = setupvalue
	if not debug or #debug == 0 then
		debug = lines.framePlayerList
	end
	if not debug or #debug == 0 then
		debug = flush:GetPlayers()
	end
	for lines, setupvalue in ipairs(debug) do
		if setupvalue == localPlayer or not write(setupvalue, float) then
			continue
		end
		local parse = setupvalue.Character
		if not parse or not dofile(parse) then
			continue
		end
		local debug = path(parse)
		if not debug then
			continue
		end
		local Factory = (newpath and parse:FindFirstChild(newpath)) or debug
		local popen, rawset = xpcall:WorldToViewportPoint(debug.Position)
		if not rawset then
			continue
		end
		if number and not seek_searchpath(Factory) then
			continue
		end
		local rawget = ((Vector2.new(popen.X, popen.Y) - searchpath)).Magnitude
		if rawget > char then
			continue
		end
		local create = settings.TargetPriority == "Lowest HP"
		local load = create and parse:FindFirstChildOfClass("Humanoid") or nil
		readline[#readline + 1] = {
			part = Factory,
			screenDist = rawget,
			worldDist = isLinux and ((debug.Position - isLinux.Position)).Magnitude or rawget,
			health = load and load.Health or math.huge,
		}
	end
	pcall_searchpath(readline)
	return readline
end

	if not searchpath then
		return
	end

	local float = char.Shoot or char.shoot

		xpcall("[Octopus Hub] GunHandler has no Shoot/shoot function.")
		return
	end
	if raise_searchpath then
		return
	end
	local setupvalue = float
	local parse = newcclosure(function(...)
		local searchpath = select(1, ...)
		if checkcaller() then
			return setupvalue(...)
		end
		if not settings.SilentAim and not settings.AnonymousBullet then
			return setupvalue(...)
		end
		if type(searchpath) ~= "table" then
			return setupvalue(...)
		end
		local char = searchpath
		if not getupvalue_searchpath(char.Shooter) then
			return setupvalue(...)
		end
		local float
		if settings.SilentAim then
			local lines = arr_searchpath()
			if lines then
				local searchpath = lines.Parent
				float = (searchpath and searchpath:FindFirstChild(settings.TargetBone)) or lines
			end
		end

			float = char.Hit
		end

			local lines = include()

				local searchpath = RaycastParams.new()
				searchpath.FilterType = Enum.RaycastFilterType.Exclude
				searchpath.FilterDescendantsInstances = { localPlayer.Character }
				local char = workspace:Raycast(
					lines.CFrame.Position,
					lines.CFrame.LookVector * 400,
					searchpath
				)

					local lines = char.Instance:FindFirstAncestorOfClass("Model")
					if lines and lines:FindFirstChildOfClass("Humanoid") then
						float = lines:FindFirstChild(settings.TargetBone) or char.Instance
					else
						float = char.Instance
					end
				end
			end
		end
		if settings.SilentAim or settings.AnonymousBullet then
			if typeof(float) == "Instance" and float.Parent then
				char.Hit = float
				char.AimPosition = float.Position
				char.Normal = Vector3.new(0, 1, 0)
				if settings.AnonymousBullet then
					local searchpath = float:FindFirstAncestorOfClass("Model") or float.Parent

					if typeof(setupvalue) == "Vector3" then
						char.ForcedOrigin = setupvalue
						lines.lastAnonOrigin = setupvalue
					end
				end

					local lines = searchpath:FindFirstChild("Muzzle")
					float = (lines and lines.WorldPosition) or searchpath.Position
				end
				local setupvalue = char.AimPosition

					local lines = setupvalue - float
					if lines.Magnitude > 20 then
						setupvalue = float + lines.Unit * 20
					end
				elseif typeof(float) == "Vector3" then
					setupvalue = float
				end
				if typeof(setupvalue) == "Vector3" then
					local searchpath = obj(setupvalue, nil)
					if typeof(searchpath) == "Vector3" then
						char.ForcedOrigin = searchpath
						lines.lastAnonOrigin = searchpath
					end
				end
			end
		end
		return setupvalue(...)
	end)
	if hookfunction then
		local lines, searchpath = pcall(function()
			return hookfunction(float, parse)
		end)

			setupvalue = searchpath
		end
	end

	end

			char.Shoot = setupvalue
		end
		if char.shoot == parse then
			char.shoot = setupvalue
		end

local function rand_searchpath(lines, searchpath, char)
	if type(lines) ~= "table" or not (searchpath and searchpath.Parent) then
		return false
	end
	local float = math.floor(tonumber(char) or #lines)
	if i() then
		float = math.clamp(float, 1, 8)
	elseif float < 1 then
		float = math.max(1, #lines)
	end
	local setupvalue = searchpath.Position
	local parse = Vector3.new(0, 1, 0)
	for char = 1, float, 1 do
		local float = lines[char]
		if type(float) ~= "table" then
			float = {}
			lines[char] = float
		end
		float.AimPosition = setupvalue
		float.RawAimPosition = setupvalue
		float.Result1 = setupvalue
		float.Result2 = searchpath
		if float.Result3 == nil then
			float.Result3 = parse
		end
		float.AimLockPart = searchpath
	end
	for searchpath = #lines, float + 1, -1 do
		lines[searchpath] = nil
	end
	return float > 0
end
local function os_searchpath(lines, searchpath)
	if not i() then
		return
	end
	local char = lines and lines.Parent
	if not (char and (char:IsA("Tool") and searchpath)) then
		return
	end
	local float = 0.3
	if size and size.getGunCooldown then
		float = size.getGunCooldown(char)
	end
	task.delay(float, function()
		if not settings.InstantKill or bytecode.active then
			return
		end
		if not (searchpath and (searchpath.Parent and char.Parent)) then
			return
		end
		local lines = char:FindFirstChild("Ammo")
		if lines and lines.Value <= 0 then
			return
		end
		if not (size and size.fireAtPart) then
			return
		end
		size.fireAtPart(
			searchpath,
			{ gun = char, silent = true, skipBusy = true, skipEquip = true, skipAmmoReload = true }
		)
	end)
end

local function idx_searchpath(lines)
	if not j_searchpath(lines) then
		return nil
	end
	local searchpath = lines[4]
	local char = type(searchpath) == "table" and #searchpath or 0
	if i() and char >= 1 then
		local char = tostring_searchpath(searchpath, lines[6])
		if char then
			local float = lines[2] and lines[2].Parent
			local setupvalue = gethook()
			if size and size.resolveBulletCount then
				setupvalue = size.resolveBulletCount(float, {})
			end
			pcall(rand_searchpath, searchpath, char, setupvalue)
			os_searchpath(lines[2], char)
			return "passthrough"
		end
	end
	local float, setupvalue = pcall(string_searchpath, lines[2], lines[4], lines[6])
	if float and setupvalue then
		if i() then
			os_searchpath(lines[2], tostring_searchpath(lines[4], lines[6]))
		end
		return "consumed"
	end
	return nil
end
local function code_searchpath(searchpath, char)
	if char or not settings.AnonymousBullet or searchpath[1] ~= "ShootGun" then
		return false
	end
	local float = lines.lastAnonOrigin
	if typeof(float) ~= "Vector3" then
		float = package(searchpath[4], searchpath[6])
	end
	if typeof(float) ~= "Vector3" then
		return false
	end
	searchpath[3] = float
	lines.lastAnonOrigin = nil
	return true
end

local function getlocal_searchpath()
	local searchpath
	local char
	local float
	local setupvalue
	local parse
	local function isLinux(lines, searchpath)
		if
			typeof(searchpath) == "EnumItem"
			and (searchpath.EnumType == Enum.KeyCode or searchpath.EnumType == Enum.UserInputType)
		then
			settings[lines] = searchpath
		end
	end
	local function xpcall(lines, searchpath)
		local char = lines:Get()
		if type(char) == "table" and (char[1] and (char[2] and Enum[char[1]])) then
			local lines = Enum[char[1]][char[2]]
			if lines then
				settings[searchpath] = lines
			end
		end
	end
	parse = function()
		for lines, searchpath in pairs(void) do
			if type(searchpath.Get) == "function" then
				local char, float = pcall(function()
					return searchpath:Get()
				end)
				if char and settings[lines] ~= nil then
					if lines == "TriggerDelayMs" then
						settings.TriggerDelay = float / 1000
					elseif
						lines == "FOVColor"
						or lines == "HighlightFillColor"
						or lines == "HighlightOutlineColor"
						or lines == "SnapLineColor"
					then
						local searchpath = float
						if type(float) == "table" and float.Color then
							searchpath = float.Color
						end
						if typeof(searchpath) == "Color3" then
							settings[lines] = searchpath
						end
					elseif lines ~= "CFrameSpeedKey" and lines ~= "FlyKey" then
